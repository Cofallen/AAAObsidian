const today = moment(dv.current().file.name, "YYYY-MM-DD");
const yesterday = today.clone().subtract(1, "day");

// ===== 昨天页面 =====
const yPage = dv.page(yesterday.format("YYYY-MM-DD"));

// ===== 昨天 score（唯一依据）=====
let yScore = yPage?.score ?? null;

// ===== 状态判断（只看昨天）=====
let state = "start";

if (yScore !== null) {
    if (yScore >= 60) state = "good";
    else if (yScore >= 30) state = "mid";
    else state = "bad";
}

// ===== 读取语录文件 =====
const file = app.vault.getAbstractFileByPath("Templates/ZZZCompare.md");
const raw = await app.vault.read(file);

// ===== 解析函数 =====
function parseSection(name) {
    const match = raw.match(new RegExp(name + ":[\\s\\S]*?(?=\\n\\w+:|$)"));
    if (!match) return [];
    return match[0]
        .replace(name + ":", "")
        .split("\n")
        .map(l => l.replace(/^-\\s*/, "").trim())
        .filter(Boolean);
}

// ===== 映射（启动语 + 昨日评价）=====
const messages = {
    start: parseSection("start"),
    good: parseSection("better"),
    mid: parseSection("mid"),
    bad: parseSection("worse")
};

const pick = arr => arr[Math.floor(Math.random() * arr.length)];

const startMsg = pick(messages.start);
const stateMsg = pick(messages[state] || messages.mid);

// ===== 输出 =====
dv.paragraph(`📌 ${startMsg} ${stateMsg}`);
